cmake_minimum_required(VERSION 3.20)
project(SimpleRemoteDeskNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(SimpleRemoteDesk.Native SHARED native_capture.cpp)
target_compile_definitions(SimpleRemoteDesk.Native PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(SimpleRemoteDesk.Native PRIVATE windowscodecs ole32 gdi32 user32)
if(MSVC)
  target_compile_options(SimpleRemoteDesk.Native PRIVATE /O2 /GL /EHsc /permissive- /W4)
  target_link_options(SimpleRemoteDesk.Native PRIVATE /LTCG /OPT:REF /OPT:ICF)
endif()
set_target_properties(SimpleRemoteDesk.Native PROPERTIES OUTPUT_NAME "SimpleRemoteDesk.Native")
